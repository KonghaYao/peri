use peri_acp_types::session_resources::ChildSnapshot;
use peri_acp_types::store::InheritedContext;
use peri_acp_types::system_reminder::{

    ReminderSource, SystemReminder, TrustedSystemReminderFactory, SYSTEM_REMINDER_VERSION,
};

/// 只有 Model 受众的 canonical reminder：`displayable_to_client()` = false。
fn s11_model_only_reminder(body: &str) -> PersistedPayload {
    let reminder = TrustedSystemReminderFactory::for_producer()
        .construct(SystemReminder {
            version: SYSTEM_REMINDER_VERSION,
            category: ReminderCategory::Task,
            source: ReminderSource("s11_probe".into()),
            kind: "probe".into(),
            severity: ReminderSeverity::Info,
            delivery: ReminderDelivery::Configurable,
            audiences: ReminderAudiences(vec![ReminderAudience::Model]),
            body: body.to_owned(),
            summary: None,
            metadata: serde_json::json!({}),
        })
        .unwrap();
    PersistedPayload::SystemReminder {
        id: MessageId::new(),
        reminder,
    }
}

/// 子会话身份 System 条目（`session/subagent/factory/spawn.rs` 的持久化形状）。
fn s11_identity(text: &str) -> PersistedPayload {
    PersistedPayload::Message(BaseMessage::system(text))
}

/// 真实 Full 形状：把**全部未排除的自有 non-System 条目**标 excluded，并追加一条摘要消息
/// （照抄 `peri-agent/src/agent/compact_v2/full.rs:86-126` 的 flag_updates 选择规则）。
async fn s11_full_like(cfg: &AcpServerConfig, sid: &str, summary: &str) -> MessageId {
    let snapshot = cfg.session_resources.load_session_snapshot(&sid.to_string()).await.unwrap();
    let flag_updates: Vec<(MessageId, MessageFlags)> = snapshot
        .payloads
        .iter()
        .filter(|payload| !snapshot.flags.get(&payload.id()).is_some_and(|f| f.excluded))
        .filter(|payload| !matches!(payload.as_message(), Some(BaseMessage::System { .. })))
        .map(|payload| {
            (
                payload.id(),
                MessageFlags {
                    excluded: true,
                    ..Default::default()
                },
            )
        })
        .collect();
    let summary_message = BaseMessage::ai(summary);
    let summary_id = summary_message.id();
    cfg.session_resources
        .apply_compaction(
            &sid.to_string(),
            &CompactionChange {
                flag_updates,
                appended_messages: vec![summary_message],
            },
        )
        .await
        .unwrap();
    summary_id
}

/// 真实子会话：`save_child` 落继承区（与 subagent spawn 的生产入口同一个 API）。
async fn s11_save_child(
    cfg: &AcpServerConfig,
    cwd: &str,
    parent: &str,
    inherited: Vec<PersistedPayload>,
) -> String {
    let snapshot = cfg
        .session_resources
        .load_session_snapshot(&parent.to_string())
        .await
        .unwrap();
    let BindingState::Bound(binding) = snapshot.binding else {
        panic!("父会话必须有执行绑定")
    };
    let FrozenState::Present(frozen) = snapshot.frozen else {
        panic!("父会话必须有已保存的 frozen 快照")
    };
    let child_id = uuid::Uuid::now_v7().to_string();
    cfg.session_resources
        .save_child(&ChildSnapshot {
            target: NewSession {
                thread_id: child_id.clone(),
                created_at: chrono::Utc::now().to_rfc3339(),
                meta: NewSessionMeta {
                    title: Some("s11 child".into()),
                    cwd: cwd.to_owned(),
                    parent_thread_id: Some(parent.to_owned()),
                    hidden: true,
                    cancel_policy: Default::default(),
                    snapshot_at_message_id: None,
                },
                binding,
                frozen,
            },
            parent_id: parent.to_owned(),
            root_id: parent.to_owned(),
            inherited: InheritedContext {
                payloads: inherited,
                flags: Default::default(),
            },
        })
        .await
        .unwrap();
    child_id
}

/// canonical 形状快照：`(有效视图, 全部自有行 id 顺序, 被排除行 id)`。
async fn s11_shape(cfg: &AcpServerConfig, sid: &str) -> (Vec<PersistedPayload>, Vec<MessageId>, Vec<MessageId>) {
    let canonical = cfg.session_resources.load_session_snapshot(&sid.to_string()).await.unwrap();
    let effective = cfg
        .session_resources
        .load_active_snapshot(&sid.to_string())
        .await
        .unwrap()
        .payloads;
    let excluded = canonical
        .payloads
        .iter()
        .filter(|payload| canonical.flags.get(&payload.id()).is_some_and(|f| f.excluded))
        .map(PersistedPayload::id)
        .collect();
    (
        effective,
        canonical.payloads.iter().map(PersistedPayload::id).collect(),
        excluded,
    )
}

/// 生产不变量：**没有任何被排除的可展示行比某个未被排除的可展示行更晚**
/// （即排除集是「最旧那一段」，锚点选出的最旧可展示条目必然比全部归档行更晚）。
/// 返回违反者，供断言直接打印。
async fn s11_exclusion_invariant_violations(cfg: &AcpServerConfig, sid: &str) -> Vec<(MessageId, MessageId)> {
    let canonical = cfg.session_resources.load_session_snapshot(&sid.to_string()).await.unwrap();
    // 方向：被排除的可展示行**不得**晚于任何未被排除的可展示行（排除集必须是最旧那一段）。
    let mut first_effective: Option<MessageId> = None;
    let mut violations = Vec::new();
    for payload in &canonical.payloads {
        let excluded = canonical
            .flags
            .get(&payload.id())
            .is_some_and(|flags| flags.excluded);
        if !payload.displayable_to_client() {
            continue;
        }
        match (excluded, first_effective) {
            (false, None) => first_effective = Some(payload.id()),
            (true, Some(effective)) => violations.push((payload.id(), effective)),
            _ => {}
        }
    }
    violations
}

/// 走完一整条页链（直到 `nextCursor = null`），返回逐页的可展示条目 id（页内历史正序）。
async fn s11_walk_chain(
    cfg: &AcpServerConfig,
    sid: &str,
    first: peri_acp_types::history_page::HistoryCursor,
    limit: usize,
) -> Vec<Vec<MessageId>> {
    let mut pages = Vec::new();
    let mut next = Some(first);
    for _ in 0..16 {
        let Some(cursor) = next else { return pages };
        let page = crate::dispatch::load_history_page(
            cfg.controller.as_ref(),
            sid,
            &cursor,
            limit,
            history_page::PAGE_BYTE_BUDGET,
        )
        .await
        .expect("页链必须可读");
        pages.push(page.items.iter().map(|item| item.id).collect());
        next = page.next_cursor;
    }
    panic!("页链没有收敛");
}

/// 判据 1 / 3（反例构造）：**有效视图里一条可展示条目都没有**，但存储里还有一条更老的
/// **可展示**归档行（h1）。兜底锚点必须是「最新一条有效行」（model-only reminder），
/// 才可能把 h1 翻出来；退回有效视图第一条（System）会直接谎称「没有更早历史」。
#[tokio::test]
#[serial]
async fn s11_probe_fallback_anchor_keeps_an_archived_row_reachable() {
    let tmp = tempfile::TempDir::new().unwrap();
    let fixture = PageFixture::new(&tmp).await;
    let sid = create_bound_fixture(
        &fixture.cfg,
        tmp.path().to_str().unwrap(),
        Some("s11-fallback-anchor"),
    )
    .await;

    let system = s11_identity("identity");
    let archived = PersistedPayload::Message(BaseMessage::human("archived body"));
    let reminder = s11_model_only_reminder("model-only note");
    fixture
        .cfg
        .session_resources
        .append_history(&sid, &[system.clone(), archived.clone(), reminder.clone()])
        .await
        .unwrap();
    // 只归档归档行（模拟「摘要被移除后仍被排除的正文」这一形状的极端版）。
    fixture
        .cfg
        .session_resources
        .apply_compaction(
            &sid,
            &CompactionChange {
                flag_updates: vec![(
                    archived.id(),
                    MessageFlags {
                        excluded: true,
                        ..Default::default()
                    },
                )],
                appended_messages: vec![],
            },
        )
        .await
        .unwrap();

    let (effective, _, _) = s11_shape(&fixture.cfg, &sid).await;
    assert_eq!(
        effective.iter().map(PersistedPayload::id).collect::<Vec<_>>(),
        vec![system.id(), reminder.id()],
        "形状前提：有效视图 = [身份 System, model-only reminder]，一条可展示条目都没有"
    );
    assert!(
        effective.iter().all(|payload| !payload.displayable_to_client()),
        "形状前提：有效视图没有任何可展示条目"
    );

    let initial = crate::dispatch::initial_history_cursor(&fixture.cfg.controller, &sid)
        .await
        .unwrap();
    let cursor = initial
        .next_cursor
        .expect("存储里还有一条更老的可展示行，不得谎称没有更早历史");
    assert_eq!(
        cursor.anchor,
        reminder.id(),
        "兜底锚点必须是**最新**一条有效行（不是有效视图第一条 System）"
    );

    let pages = s11_walk_chain(&fixture.cfg, &sid, cursor, 50).await;
    let delivered: Vec<MessageId> = pages.iter().flatten().copied().collect();
    assert_eq!(
        delivered,
        vec![archived.id()],
        "归档正文必须被页链交付（可展示 canonical 行不得不可达）"
    );
}

/// 判据 1⑤ / 2：页链里允许出现「整页不可展示」的空页，但它**不得**终止浏览：

    );
}

/// 判据 1 / 2 端到端（真实子会话形状）：继承区在前 + own 首条是身份 System + 一次 Full。
/// 断言「每一个可展示 canonical 行恰好交付一次」，且祖先条目永不进入页。
#[tokio::test]
#[serial]
async fn s11_probe_child_session_delivers_every_displayable_own_row_exactly_once() {
    let tmp = tempfile::TempDir::new().unwrap();
    let fixture = PageFixture::new(&tmp).await;
    let cwd = tmp.path().to_str().unwrap().to_owned();

    let parent = create_bound_fixture(&fixture.cfg, &cwd, Some("s11-parent")).await;
    let p1 = PersistedPayload::Message(BaseMessage::human("inherited turn 1"));
    let p2 = PersistedPayload::Message(BaseMessage::ai("inherited answer 1"));
    fixture
        .cfg
        .session_resources
        .append_history(&parent, &[p1.clone(), p2.clone()])
        .await
        .unwrap();

    let child = s11_save_child(&fixture.cfg, &cwd, &parent, vec![p1.clone(), p2.clone()]).await;
    let system = s11_identity("child identity");
    let h1 = PersistedPayload::Message(BaseMessage::human("archived question"));
    let h2 = PersistedPayload::Message(BaseMessage::ai("archived answer"));
    fixture
        .cfg
        .session_resources
        .append_history(&child, &[system.clone(), h1.clone(), h2.clone()])
        .await
        .unwrap();
    let summary = s11_full_like(&fixture.cfg, &child, "compacted summary").await;

    let snapshot = fixture
        .cfg
        .session_resources
        .load_session_snapshot(&child)
        .await
        .unwrap();
    assert_eq!(
        snapshot.inherited.payloads.iter().map(PersistedPayload::id).collect::<Vec<_>>(),
        vec![p1.id(), p2.id()],
        "形状前提：继承区冻结了父前缀"
    );
    let (effective, own_order, archived) = s11_shape(&fixture.cfg, &child).await;
    assert_eq!(
        own_order,
        vec![system.id(), h1.id(), h2.id(), summary],
        "形状前提：own 区首条是身份 System 条目"
    );
    assert_eq!(
        effective.iter().map(PersistedPayload::id).collect::<Vec<_>>(),
        vec![system.id(), summary],
        "形状前提：有效视图 = [身份 System, 摘要]"
    );
    assert!(
        archived.contains(&h1.id()) && archived.contains(&h2.id()),
        "形状前提：压缩原文已被归档"
    );

    let initial = crate::dispatch::initial_history_cursor(&fixture.cfg.controller, &child)
        .await
        .unwrap();
    let cursor = initial
        .next_cursor
        .expect("存储里有压缩原文，不得谎称没有更早历史");
    assert_eq!(cursor.anchor, summary, "锚点必须是摘要，不得落在身份 System 或祖先条目上");
    assert!(
        !snapshot
            .inherited
            .payloads
            .iter()
            .any(|payload| payload.id() == cursor.anchor),
        "锚点不得落在继承条目上（祖先条目在本线程没有行）"
    );

    // 首屏 = 有效视图里的**可展示**条目（回放出口的过滤口径）。
    let first_screen: Vec<MessageId> = effective
        .iter()
        .filter(|payload| payload.displayable_to_client())
        .map(PersistedPayload::id)
        .collect();
    let pages = s11_walk_chain(&fixture.cfg, &child, cursor, 50).await;
    let delivered: Vec<MessageId> = pages.iter().flatten().copied().collect();

    // 无漏：全部可展示 canonical 行恰好出现一次（首屏 ∪ 页 = 全部可展示行，且无交集）。
    let mut union = first_screen.clone();
    union.extend(delivered.iter().copied());
    let mut expected: Vec<MessageId> = own_order
        .iter()
        .copied()
        .filter(|id| {
            snapshot
                .payloads
                .iter()
                .find(|payload| payload.id() == *id)
                .is_some_and(PersistedPayload::displayable_to_client)
        })
        .collect();
    let mut union_sorted = union.clone();
    union_sorted.sort_by_key(|id| id.as_uuid());
    expected.sort_by_key(|id| id.as_uuid());
    assert_eq!(union_sorted, expected, "首屏 + 页必须覆盖全部可展示自有行");
    assert_eq!(
        delivered,
        vec![h1.id(), h2.id()],
        "归档正文按历史正序交付，且与首屏零重叠"
    );
    // 祖先条目永不进入页（它们在本线程没有行）。
    assert!(
        !delivered.contains(&p1.id()) && !delivered.contains(&p2.id()),
        "继承条目不得出现在页里"
    );
    assert!(
        s11_exclusion_invariant_violations(&fixture.cfg, &child)
            .await
            .is_empty(),
        "排除集必须只覆盖最旧那一段（锚点规则的前提）"
    );
}

/// 判据 1④（两轮以上 Full）+ 无重：两次 Full 之后，第一个摘要也变成归档原文，
/// 必须能被页链找回；锚点仍是首屏最后一条可展示有效条目。
#[tokio::test]
#[serial]
async fn s11_probe_two_full_rounds_keep_every_archived_row_reachable() {
    let tmp = tempfile::TempDir::new().unwrap();
    let fixture = PageFixture::new(&tmp).await;
    let cwd = tmp.path().to_str().unwrap().to_owned();
    let sid = create_bound_fixture(&fixture.cfg, &cwd, Some("s11-two-fulls")).await;

    let system = s11_identity("identity");
    let h1 = PersistedPayload::Message(BaseMessage::human("turn 1"));
    let h2 = PersistedPayload::Message(BaseMessage::ai("answer 1"));
    let h3 = PersistedPayload::Message(BaseMessage::human("turn 2"));
    fixture
        .cfg
        .session_resources
        .append_history(&sid, &[system.clone(), h1.clone(), h2.clone(), h3.clone()])
        .await
        .unwrap();
    let summary1 = s11_full_like(&fixture.cfg, &sid, "summary one").await;

    let h4 = PersistedPayload::Message(BaseMessage::human("turn 3"));
    let h5 = PersistedPayload::Message(BaseMessage::ai("answer 3"));
    fixture
        .cfg
        .session_resources
        .append_history(&sid, &[h4.clone(), h5.clone()])
        .await
        .unwrap();
    let summary2 = s11_full_like(&fixture.cfg, &sid, "summary two").await;

    let (effective, own_order, archived) = s11_shape(&fixture.cfg, &sid).await;
    assert_eq!(
        effective.iter().map(PersistedPayload::id).collect::<Vec<_>>(),
        vec![system.id(), summary2],
        "形状前提：两轮 Full 后有效视图只剩 [身份 System, 第二个摘要]"
    );
    assert_eq!(
        archived,
        vec![h1.id(), h2.id(), h3.id(), summary1, h4.id(), h5.id()],
        "形状前提：第一轮的全部历史 + 第一个摘要都在归档区"
    );
    assert!(
        s11_exclusion_invariant_violations(&fixture.cfg, &sid).await.is_empty(),
        "两轮 Full 之后排除集仍是最旧那一段"
    );

    let initial = crate::dispatch::initial_history_cursor(&fixture.cfg.controller, &sid)
        .await
        .unwrap();
    let cursor = initial.next_cursor.expect("归档区存在时必须给游标");
    assert_eq!(cursor.anchor, summary2, "锚点 = 首屏唯一可展示条目（第二个摘要）");

    let pages = s11_walk_chain(&fixture.cfg, &sid, cursor, 50).await;
    let delivered: Vec<MessageId> = pages.iter().flatten().copied().collect();
    assert_eq!(
        delivered,
        vec![h1.id(), h2.id(), h3.id(), summary1, h4.id(), h5.id()],
        "两轮归档的全部可展示行都必须交付，且第一个摘要不丢"
    );
    assert!(!delivered.contains(&summary2), "首屏条目不得在页里重复");
    assert_eq!(
        delivered.len() + 2,
        own_order.len(),
        "页 6 条 + 首屏的 summary2 + 不交付的身份 System = 全部 8 条自有行"
    );
}

/// 判据 1①（边界构造，**非生产形状**）：被排除行比「最旧可展示有效条目」更晚。
/// 只由绕过 `full.rs` 的 `apply_compaction` 直接调用构造；本探针断言这一形状下
/// 锚点之后的归档行**确实不可达**（把修复的适用边界钉死成可复现事实）。
#[tokio::test]
#[serial]
async fn s11_probe_non_prefix_exclusion_is_out_of_the_anchor_rule_scope() {
    let tmp = tempfile::TempDir::new().unwrap();
    let fixture = PageFixture::new(&tmp).await;
    let sid = create_bound_fixture(

    let h0 = PersistedPayload::Message(BaseMessage::human("archived head"));
    let h1 = PersistedPayload::Message(BaseMessage::ai("oldest effective"));
    let h2 = PersistedPayload::Message(BaseMessage::human("excluded but NEWER than h1"));
    let h3 = PersistedPayload::Message(BaseMessage::ai("newest effective"));
    fixture
        .cfg
        .session_resources
        .append_history(&sid, &[h0.clone(), h1.clone(), h2.clone(), h3.clone()])
        .await
        .unwrap();
    let archived_flag = MessageFlags {
        excluded: true,
        ..Default::default()
    };
    fixture
        .cfg
        .session_resources
        .apply_compaction(
            &sid,
            &CompactionChange {
                flag_updates: vec![
                    (h0.id(), archived_flag.clone()),
                    (h2.id(), archived_flag.clone()),
                ],
                appended_messages: vec![],
            },
        )
        .await
        .unwrap();

    let (effective, _, excluded) = s11_shape(&fixture.cfg, &sid).await;
    assert_eq!(
        effective.iter().map(PersistedPayload::id).collect::<Vec<_>>(),
        vec![h1.id(), h3.id()],
        "形状前提：有效视图 = [h1, h3]"
    );
    assert_eq!(excluded, vec![h0.id(), h2.id()], "形状前提：h0 与 h2 被排除");
    assert!(
        !s11_exclusion_invariant_violations(&fixture.cfg, &sid)
            .await
            .is_empty(),
        "这个形状确实违反生产不变量（排除集不再是「最旧那一段」）"
    );

    let initial = crate::dispatch::initial_history_cursor(&fixture.cfg.controller, &sid)
        .await
        .unwrap();
    let cursor = initial
        .next_cursor
        .expect("更早方向还有 h0 一行，必须给游标");
    assert_eq!(cursor.anchor, h1.id(), "锚点 = 最旧可展示有效条目（h0 之后）");

    let pages = s11_walk_chain(&fixture.cfg, &sid, cursor, 50).await;
    let delivered: Vec<MessageId> = pages.iter().flatten().copied().collect();
    let first_screen = vec![h1.id(), h3.id()];
    assert_eq!(delivered, vec![h0.id()], "页只覆盖锚点之前的归档行");
    assert!(
        !delivered.contains(&h2.id()) && !first_screen.contains(&h2.id()),
        "边界事实：比锚点更晚的归档行 h2 既不在首屏、也不被任何一页交付（漏项）"
    );
}

/// 判据 1② / 2（真实 rewind API）：rewind 之后仍被排除的行 + 仍有效的行并存，
/// 锚点必须落在「仍有效的最旧可展示条目」，归档行按历史正序交付一次，
/// 且热态（首屏来源）与存储有效视图（锚点来源）必须是同一份事实。
#[tokio::test]
#[serial]
async fn s11_probe_rewind_keeps_anchor_and_first_screen_on_the_same_facts() {
    use crate::host::history_page::HistoryPageRequest;
    use peri_acp_types::history_page::HistoryCursorEnvelope;

    let tmp = tempfile::TempDir::new().unwrap();
    let peri_config = make_peri_config_with_provider(make_provider_config(
        "a",
        "openai",
        "sk-openai-test",
        "gpt-4o",
    ));
    let provider = LlmProvider::from_config(&peri_config).unwrap();
    let cfg = make_server_config(peri_config, provider, &tmp).await;
    let transport: Arc<dyn crate::transport::AcpTransport> = Arc::new(MockTransport::default());
    let shared: crate::host::SharedSessions = Arc::new(tokio::sync::Mutex::new(HashMap::new()));
    let sid = {
        let mut sessions = shared.lock().await;
        register_session_with_history(&mut sessions, tmp.path().to_str().unwrap(), &cfg).await
    };
    // 历史 = [h1, a1, h2]；追加第 4 条 tail（存储 + 热态同源）。
    let (h1, a1, h2) = {
        let s = shared.lock().await;
        let state = s.get(&sid).unwrap();
        (state.history[0].id(), state.history[1].id(), state.history[2].id())
    };
    let tail = BaseMessage::human("第四轮用户问题");
    cfg.session_resources
        .append_history(
            &sid,
            &[PersistedPayload::Message(tail.clone())],
        )
        .await
        .unwrap();
    {
        let mut sessions = shared.lock().await;
        let state = sessions.get_mut(&sid).unwrap();
        state.history_payloads.push(PersistedPayload::Message(tail.clone()));
        state.history.push(tail.clone());
    }
    // 归档前两条（存储），热态同步成有效视图 [h2, tail]。
    let excluded = MessageFlags {
        excluded: true,
        ..Default::default()
    };
    cfg.session_resources
        .apply_message_projections(&sid, &[(h1, excluded.clone()), (a1, excluded.clone())])
        .await
        .unwrap();
    {
        let mut sessions = shared.lock().await;
        let state = sessions.get_mut(&sid).unwrap();
        let kept: Vec<PersistedPayload> = state
            .history_payloads
            .iter()
            .filter(|payload| payload.id() != h1 && payload.id() != a1)
            .cloned()
            .collect();
        state.history_payloads = kept.clone();
        state.history = kept
            .iter()
            .filter_map(|payload| payload.as_message().cloned())
            .collect();
    }
    cfg.session_manager.caps_registry().insert(
        sid.clone(),
        PeriCaps {
            rewind: true,
            ..PeriCaps::default()
        },
    );

    let target = tail.id().as_uuid().to_string();
    let preview = {
        let mut sessions = shared.lock().await;
        handle_request(
            "session/rewind-preview",
            &json!({ "sessionId": sid, "target_message_id": target }),
            &cfg,
            &mut sessions,
            &transport,
        )
        .await
        .unwrap()
    };
    let fingerprint = preview["preview_fingerprint"].as_str().unwrap().to_string();
    {
        let mut sessions = shared.lock().await;
        handle_request(
            "session/rewind",
            &json!({
                "sessionId": sid,
                "target_message_id": target,
                "preview_fingerprint": fingerprint,
            }),
            &cfg,
            &mut sessions,
            &transport,
        )
        .await
        .unwrap();
    }

    // 热态（首屏来源）与存储有效视图（锚点来源）必须一致。
    let hot: Vec<MessageId> = shared
        .lock()
        .await
        .get(&sid)
        .unwrap()
        .history_payloads
        .iter()
        .map(PersistedPayload::id)
        .collect();
    let (effective, _, still_excluded) = s11_shape(&cfg, &sid).await;
    assert_eq!(
        hot,
        effective.iter().map(PersistedPayload::id).collect::<Vec<_>>(),
        "rewind 之后热态与存储有效视图必须是同一份事实"
    );
    assert_eq!(hot, vec![h2], "形状前提：有效视图只剩活着的摘要行 h2");
    assert_eq!(
        still_excluded,
        vec![h1, a1],
        "形状前提：前两条仍被排除（摘要活着）"
    );
    assert!(
        s11_exclusion_invariant_violations(&cfg, &sid).await.is_empty(),
        "rewind 之后排除集仍是最旧那一段"
    );

    let initial = crate::dispatch::initial_history_cursor(&cfg.controller, &sid)
        .await
        .unwrap();
    let cursor = initial.next_cursor.expect("归档区存在时必须给游标");
    assert_eq!(cursor.anchor, h2, "锚点 = 最旧可展示有效条目（活着的摘要）");

    let pagination = PeriCaps {
        history_pagination: true,
        ..PeriCaps::default()
    };
    let envelope = HistoryCursorEnvelope::new(
        shared.lock().await.get(&sid).unwrap().structure_version,
        "s11-rewind-snapshot",
        cursor,
    );
    let mut ids: Vec<String> = Vec::new();
    let mut next = Some(envelope);
    for _ in 0..8 {
        let Some(current) = next else { break };
        let page = crate::host::history_page::handle_page(
            &cfg,
            &shared,
            &pagination,
            &HistoryPageRequest {
                session_id: sid.clone(),
                cursor: current,
                limit: 50,
            },
        )
        .await
        .unwrap();
        for item in page["items"].as_array().unwrap() {
            ids.push(item["entryId"].as_str().unwrap().to_owned());
        }
        next = page["nextCursor"]
            .as_str()
            .and_then(|raw| HistoryCursorEnvelope::decode(raw, &sid).ok());
    }
    assert_eq!(
        ids,
        vec![crate::dispatch::entry_id(&sid, h1), crate::dispatch::entry_id(&sid, a1)],
        "归档行必须按历史正序交付一次：{ids:?}"
    );
    assert!(
        !ids.contains(&crate::dispatch::entry_id(&sid, h2)),
        "首屏已展示的摘要不得在页里重复"
    );
}
