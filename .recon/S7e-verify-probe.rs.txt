//! S7e 复核者探针（临时文件，复核结束即删，工作区不留痕）。
//!
//! 目的（全部自己构造，不采信实现者用例）：
//! 1. 端到端真实发布路径（`handle_history_page_loaded` → `push_view_models` →
//!    `Transcript::prepare`）下，桥接槽位数 ≠ 渲染槽位数时，**同一可见单元屏幕行**是否不变；
//! 2. 场景 A（插入在视口上方）/ 场景 B（中间缺口填补）是否都不跳；
//! 3. 边界：跟随态、`usize::MAX` 哨兵、无并合帧；
//! 4. 残留缺口：锚点单元自身被并入新折叠组而消失时发生了什么（量化）。
//!
//! 度量口径与生产 effect 完全一致：`屏幕行 = slot_visual_start(slot) − 视口偏移`，
//! 视口偏移 = `PrependRelocation::target_for(false, scroll_y)`（`None` → offset 不动，
//! 与 `auto_follow.rs` 的 `if let Some(...)` 分支同义）。
//! 单元定位**按内容自行查找**（不采信实现者用例里的槽位常量）。

fn context() -> EntryRenderContext {
    crate::kit::atoms::init_atoms();
    crate::i18n::init(Some("en"));
    EntryRenderContext {
        theme: Arc::clone(&THEME_ATOM.state().read()),
        language: crate::kit::atoms::LANG_VERSION.get(),
        surface: EntrySurface::Message,
        occurrence: 0,
    }
}

fn sub(position: u64, update: serde_json::Value) -> HistorySubItem {
    HistorySubItem {
        position,
        update,
        system_reminder: false,
    }
}

fn page(
    token: &str,
    cursor: Option<&str>,
    items: Vec<(&str, u64, Vec<HistorySubItem>)>,
) -> HistoryPageResponse {
    HistoryPageResponse {
        session_id: "s1".to_string(),
        snapshot_token: token.to_string(),
        next_cursor: cursor.map(str::to_owned),
        items: items
            .into_iter()
            .map(|(entry_id, position, sub_items)| HistoryPageItem {
                entry_id: entry_id.to_string(),
                position,
                oversize: false,
                sub_items,
            })
            .collect(),
    }
}

fn user_update(text: &str) -> serde_json::Value {
    json!({
        "sessionUpdate": "user_message_chunk",
        "content": { "type": "text", "text": text },
        "messageId": "m",
    })
}

fn tool_start(tool_id: &str, tool_name: &str) -> serde_json::Value {
    json!({
        "sessionUpdate": "tool_call",
        "toolCallId": tool_id,
        "title": tool_name,
        "rawInput": { "file_path": "/tmp/x" },
    })
}

fn tool_end(tool_id: &str, output: &str, is_error: bool) -> serde_json::Value {
    json!({
        "sessionUpdate": "tool_call_update",
        "toolCallId": tool_id,
        "rawOutput": output,
        "status": if is_error { "failed" } else { "completed" },
    })
}

fn browsing_state(token: &str) -> BridgeState {
    crate::kit::atoms::init_atoms();
    history_paging::reset_browsing_state();
    *crate::kit::atoms::ACTIVE_SESSION_ID.state().write() = "s1".to_string();
    let mut state = make_fold_test_state();
    state.active_session_id = "s1".to_string();
    history_paging::note_initial_cursor("s1", Some(token.to_string()), Some("cursor-1".to_string()));
    state
}

fn seed_inherited_prefix(state: &mut BridgeState, text: &str) {
    state.committed = im::Vector::from(vec![TuiRenderUnit::TuiUserBubble(TuiUserBubble::new(
        text.to_string(),
    ))]);
}

fn seed_own_entry_via(
    state: &mut BridgeState,
    entry_id: &str,
    position: u64,
    push_unit: impl FnOnce(&mut BridgeState),
) {
    dispatch_and_notify(
        state,
        &AcpEventData::ReplayEntryMarked {
            entry_id: entry_id.to_string(),
            position,
        },
    );
    push_unit(state);
}

fn seed_own_bubble(state: &mut BridgeState, entry_id: &str, position: u64, text: &str) {
    seed_own_entry_via(state, entry_id, position, |state| {
        dispatch_and_notify(
            state,
            &AcpEventData::ReplayedUserBubble {
                input_id: format!("input-{entry_id}"),
                text: text.to_string(),
            },
        );
    });
}

fn seed_own_card(state: &mut BridgeState, entry_id: &str, position: u64, tool_id: &str, name: &str) {
    seed_own_entry_via(state, entry_id, position, |state| {
        dispatch_and_notify(
            state,
            &AcpEventData::ReplayToolStarted {
                tool_id: tool_id.to_string(),
                tool_name: name.to_string(),
                input_summary: "x".to_string(),
                raw_input: json!({"file_path": "x"}),
            },
        );
        dispatch_and_notify(
            state,
            &AcpEventData::ReplayToolEnded {
                tool_id: tool_id.to_string(),
                output_summary: "ok".to_string(),
                is_error: false,
            },
        );
    });
}

fn dispatch_landing_page(state: &mut BridgeState, token: &str, response: HistoryPageResponse) {
    {
        let handle = crate::kit::atoms::HISTORY_PAGING.state();
        let mut paging = handle.write();
        paging.session_id = "s1".to_string();
        paging.snapshot_token = Some(token.to_string());
    }
    assert!(
        crate::kit::atoms::HISTORY_PAGING
            .state()
            .read()
            .accepts("s1", token),
        "浏览身份在派发前被并行用例改写"
    );
    dispatch_and_notify(
        state,
        &AcpEventData::HistoryPageLoaded {
            response: Box::new(response),
        },
    );
}

/// 按内容自行定位单元槽位（不采信任何手工槽位常量）。
fn slot_of_user_text(index: &SlotIndex, text: &str) -> usize {
    let items = index.canonical_items().expect("索引有 items");
    items
        .iter()
        .position(|vm| matches!(vm, TuiRenderUnit::TuiUserBubble(b) if b.text == text))
        .unwrap_or_else(|| {
            let kinds: Vec<String> = items
                .iter()
                .map(|vm| match vm {
                    TuiRenderUnit::TuiUserBubble(b) => format!("bubble({})", b.text),
                    TuiRenderUnit::TuiToolCard(c) => format!("card({})", c.tool_id),
                    TuiRenderUnit::TuiCollapsedGroup(g) => format!("GROUP({})", g.view_models.len()),
                    other => format!("{other:?}").chars().take(30).collect(),
                })
                .collect();
            panic!("找不到单元 bubble({text})；当前槽位序列 = {kinds:?}")
        })
}

fn kinds(index: &SlotIndex) -> Vec<String> {
    index
        .canonical_items()
        .expect("索引有 items")
        .iter()
        .map(|vm| match vm {
            TuiRenderUnit::TuiUserBubble(b) => format!("bubble({})", b.text),
            TuiRenderUnit::TuiToolCard(c) => format!("card({})", c.tool_id),
            TuiRenderUnit::TuiCollapsedGroup(g) => format!("GROUP({})", g.view_models.len()),
            other => format!("{other:?}").chars().take(30).collect(),
        })
        .collect()
}

struct Probe {
    before: Arc<SlotIndex>,
    after: Arc<SlotIndex>,
    scroll_y: usize,
    relocation: Option<PrependRelocation>,
    bridge_slots: usize,
    anchor_text: String,
}

impl Probe {
    /// 生效的视口偏移：`target_for` 有值 → 落 offset（浏览态 effect），否则 offset 不动。
    fn applied_offset(&self) -> usize {
        match self.relocation {
            Some(relocation) => relocation
                .target_for(false, self.scroll_y)
                .expect("浏览态 + 非哨兵必须换算"),
            None => self.scroll_y,
        }
    }

    /// 锚点单元（按内容查找）在屏幕上的行号。
    fn anchored_screen_row(&self) -> (isize, isize) {
        let before_slot = slot_of_user_text(&self.before, &self.anchor_text);
        let after_slot = slot_of_user_text(&self.after, &self.anchor_text);
        let row_before = self
            .before
            .slot_visual_start(before_slot)
            .expect("before 起点") as isize
            - self.scroll_y as isize;
        let row_after = self
            .after
            .slot_visual_start(after_slot)
            .expect("after 起点") as isize
            - self.applied_offset() as isize;
        (row_before, row_after)
    }
}

/// 跑一轮真实发布：首屏发布 → `prepare` → 并入一页发布 → `prepare`。
/// 锚点 = `anchor_text` 那条用户气泡（视口顶行落在其起始行 +1 行）。
fn publish_then_relocate(
    state: &mut BridgeState,
    token: &str,
    anchor_text: &str,
    response: HistoryPageResponse,
) -> Probe {
    let grid = GridSpec::grid_for(80);
    let context = context();
    let reset = crate::kit::atoms::BRIDGE_RESET_COUNTER.get();
    let mut transcript = Transcript::default();
    let mut caches: Vec<VmCacheSlot> = Vec::new();

    let first = crate::kit::atoms::VIEW_MODELS.state().read().clone();
    let before = transcript.prepare(
        &first,
        &crate::kit::atoms::TRANSCRIPT_PUBLICATION.get(),
        &mut caches,
        grid,
        grid.line_width(),
        context.clone(),
        0,
        reset,
        0,
        10,
        crate::kit::atoms::HISTORY_PREPEND_EPOCH.get(),
    );
    let anchor_slot = slot_of_user_text(&before, anchor_text);
    let scroll_y = before.slot_visual_start(anchor_slot).expect("锚点单元起始") + 1;
    let epoch_before = crate::kit::atoms::HISTORY_PREPEND_EPOCH.get();
    assert!(
        transcript
            .take_prepend_relocation()
            .is_none(),
        "首帧（transcript 冷启动）不得产出重定位"
    );

    dispatch_landing_page(state, token, response);
    assert!(
        crate::kit::atoms::HISTORY_PREPEND_EPOCH.get() > epoch_before,
        "并入成功必须推进并合代数"
    );

    let second = crate::kit::atoms::VIEW_MODELS.state().read().clone();
    assert_ne!(second.generation, first.generation, "并入必须产生新发布");
    let after = transcript.prepare(
        &second,
        &crate::kit::atoms::TRANSCRIPT_PUBLICATION.get(),
        &mut caches,
        grid,
        grid.line_width(),
        context,
        0,
        reset,
        scroll_y,
        10,
        crate::kit::atoms::HISTORY_PREPEND_EPOCH.get(),
    );
    let relocation = transcript.take_prepend_relocation();
    Probe {
        before,
        after,
        scroll_y,
        relocation,
        bridge_slots: state.committed.len(),
        anchor_text: anchor_text.to_string(),
    }
}

// ── 判据 1：S7d 折叠组反例形状 ────────────────────────────────────────────
#[test]
#[serial_test::serial]
fn zz_s7e_collapsed_group_shape_keeps_anchor_screen_row() {
    let mut state = browsing_state("tok");
    seed_inherited_prefix(&mut state, "inherited");
    seed_own_bubble(&mut state, "s1:own", 100, "first-screen-own");
    let tools = page(
        "tok",
        Some("cursor-2"),
        vec![
            ("s1:old-1a", 10, vec![sub(0, tool_start("t1", "Read"))]),
            ("s1:old-1b", 11, vec![sub(0, tool_end("t1", "ok-1", false))]),
            ("s1:old-2a", 20, vec![sub(0, tool_start("t2", "Grep"))]),
            ("s1:old-2b", 21, vec![sub(0, tool_end("t2", "ok-2", false))]),
        ],
    );

    let probe = publish_then_relocate(&mut state, "tok", "first-screen-own", tools);
    let (row_before, row_after) = probe.anchored_screen_row();
    println!(
        "S7E-PROBE[collapsed]: 桥接槽位={} 渲染槽位={} 渲染序列={:?} before_rows={} after_rows={} scroll_y={} relocation={:?} row_before={} row_after={}",
        probe.bridge_slots,
        probe.after.slot_count(),
        kinds(&probe.after),
        probe.before.total_visual(),
        probe.after.total_visual(),
        probe.scroll_y,
        probe.relocation.map(|r| (r.from_offset, r.target_offset)),
        row_before,
        row_after,
    );
    assert_ne!(
        probe.bridge_slots,
        probe.after.slot_count(),
        "本形状必须让两把尺子长度不同（桥接 4 槽 vs 渲染 3 槽）"
    );
    assert_eq!(
        row_before, row_after,
        "折叠组形态：同一可见单元屏幕行不变（A20）"
    );
}

// ── 判据 5 边界：跟随态 / 哨兵 / 无并合帧 ────────────────────────────────
#[test]
#[serial_test::serial]
fn zz_s7e_boundaries_follow_and_sentinel_and_no_publish() {
    // 决策层：跟随态（follow_bottom = true → target_for(true, ..)）与 usize::MAX 哨兵。
    let relocation = PrependRelocation {
        from_offset: 5,
        target_offset: 42,
    };
    assert_eq!(relocation.target_for(true, 5), None, "跟随态不换算");
    assert_eq!(relocation.target_for(false, usize::MAX), None, "哨兵不换算");
    assert_eq!(
        relocation.target_for(false, 5),
        Some(42),
        "浏览态按相对量换算：scroll_y + (target − from)"
    );
    assert_eq!(
        relocation.target_for(false, 7),
        Some(44),
        "相对量口径（期间发生的其它滚动不被覆盖）"
    );

    // 无并合帧：同一快照再次 prepare → 不产出重定位（同一次发布不重复补偿）。
    let mut state = browsing_state("tok");
    seed_inherited_prefix(&mut state, "inherited");
    seed_own_bubble(&mut state, "s1:own", 100, "first-screen-own");
    let grid = GridSpec::grid_for(80);
    let context = context();
    let reset = crate::kit::atoms::BRIDGE_RESET_COUNTER.get();
    let mut transcript = Transcript::default();
    let mut caches: Vec<VmCacheSlot> = Vec::new();
    let snapshot = crate::kit::atoms::VIEW_MODELS.state().read().clone();
    for _ in 0..2 {
        transcript.prepare(
            &snapshot,
            &crate::kit::atoms::TRANSCRIPT_PUBLICATION.get(),
            &mut caches,
            grid,
            grid.line_width(),
            context.clone(),
            0,
            reset,
            0,
            10,
            crate::kit::atoms::HISTORY_PREPEND_EPOCH.get(),
        );
        assert!(
            transcript.take_prepend_relocation().is_none(),
            "同一快照的重复 prepare 不得产出重定位（无并合 / 无发布变化）"
        );
    }
}

// ── 残留缺口探针：锚点单元自身被并入新折叠组 ──────────────────────────────
#[test]
#[serial_test::serial]
fn zz_s7e_anchor_absorbed_into_new_group() {
    let mut state = browsing_state("tok");
    // 首屏：一张成功工具卡（视口顶行所在单元）+ 下方一条自有气泡。
    seed_own_card(&mut state, "s1:c1", 10, "t-anchor", "Read");
    seed_own_bubble(&mut state, "s1:own", 100, "below-own");
    // 更早一页：另一张**成功通用**工具卡（position 5/6，起点终点各一条目——生产形状），
    // 并入后紧邻窗口里的锚点卡 → 相邻成功工具卡压成一张折叠组。
    let prev = page(
        "tok",
        Some("cursor-2"),
        vec![
            ("s1:prev-a", 5, vec![sub(0, tool_start("t-prev", "Grep"))]),
            ("s1:prev-b", 6, vec![sub(0, tool_end("t-prev", "ok-prev", false))]),
        ],
    );

    let grid = GridSpec::grid_for(80);
    let context = context();
    let reset = crate::kit::atoms::BRIDGE_RESET_COUNTER.get();
    let mut transcript = Transcript::default();
    let mut caches: Vec<VmCacheSlot> = Vec::new();

    let first = crate::kit::atoms::VIEW_MODELS.state().read().clone();
    let before = transcript.prepare(
        &first,
        &crate::kit::atoms::TRANSCRIPT_PUBLICATION.get(),
        &mut caches,
        grid,
        grid.line_width(),
        context.clone(),
        0,
        reset,
        0,
        10,
        crate::kit::atoms::HISTORY_PREPEND_EPOCH.get(),
    );
    let before_kinds = kinds(&before);
    let card_slot_before = before_kinds
        .iter()
        .position(|k| k.starts_with("card("))
        .expect("首屏有工具卡");
    // 视口顶行 = 锚点卡第一行 + 0 行（锚点就是这个单元本身）。
    let scroll_y = before.slot_visual_start(card_slot_before).expect("卡起点");
    let below_row_before = before
        .slot_visual_start(slot_of_user_text(&before, "below-own"))
        .expect("气泡起点") as isize
        - scroll_y as isize;

    dispatch_landing_page(&mut state, "tok", prev);
    let second = crate::kit::atoms::VIEW_MODELS.state().read().clone();
    let after = transcript.prepare(
        &second,
        &crate::kit::atoms::TRANSCRIPT_PUBLICATION.get(),
        &mut caches,
        grid,
        grid.line_width(),
        context,
        0,
        reset,
        scroll_y,
        10,
        crate::kit::atoms::HISTORY_PREPEND_EPOCH.get(),
    );
    let relocation = transcript.take_prepend_relocation();
    let after_kinds = kinds(&after);
    let below_row_after = after
        .slot_visual_start(slot_of_user_text(&after, "below-own"))
        .expect("气泡起点") as isize
        - relocation
            .and_then(|r| r.target_for(false, scroll_y))
            .unwrap_or(scroll_y) as isize;

    println!(
        "S7E-PROBE[absorbed]: 桥接槽位={} 渲染槽位={} before={:?} after={:?} relocation={:?} 下方气泡行 before={} after={}（锚点卡高={:?} 组高={:?}）",
        state.committed.len(),
        after.slot_count(),
        before_kinds,
        after_kinds,
        relocation.map(|r| (r.from_offset, r.target_offset)),
        below_row_before,
        below_row_after,
        before.slot_visual_range(card_slot_before),
        after.slot_visual_range(0),
    );
    assert!(
        before_kinds.iter().any(|k| k.starts_with("card(t-anchor)")),
        "锚点卡在发布前是独立单元"
    );
    assert!(
        after_kinds.iter().any(|k| k.starts_with("GROUP(")),
        "锚点卡在发布后被并入折叠组（身份消失）"
    );
    assert!(
        relocation.is_none(),
        "锚点身份并入折叠组后认不出 → 按设计不补偿（安全方向）"
    );
    println!(
        "S7E-PROBE[absorbed-delta]: 视口未补偿 → 下方气泡屏幕行 {below_row_before} → {below_row_after}（位移 {} 行）",
        below_row_after - below_row_before
    );
}

// ── 判据 1 反向：插入把已有折叠组**拆开**（渲染槽位变多）──────────────────
#[test]
#[serial_test::serial]
fn zz_s7e_group_split_keeps_anchor_below_row() {
    let mut state = browsing_state("tok");
    // 首屏：两条自有工具条目（各一张成功卡，相邻）+ 一条自有气泡 → 渲染侧成组。
    seed_own_card(&mut state, "s1:c1", 10, "t1", "Read");
    seed_own_card(&mut state, "s1:c2", 20, "t2", "Grep");
    seed_own_bubble(&mut state, "s1:own", 100, "below-own");
    // 更早一页的条目落在两张卡之间（position 15）→ 组被拆开。
    let split = page(
        "tok",
        Some("cursor-2"),
        vec![("s1:mid", 15, vec![sub(0, user_update("between"))])],
    );

    let probe = publish_then_relocate(&mut state, "tok", "below-own", split);
    let (row_before, row_after) = probe.anchored_screen_row();
    println!(
        "S7E-PROBE[group-split]: 桥接槽位={} 渲染槽位={} 前索引槽位={} 序列={:?} scroll_y={} relocation={:?} row_before={} row_after={}",
        probe.bridge_slots,
        probe.after.slot_count(),
        probe.before.slot_count(),
        kinds(&probe.after),
        probe.scroll_y,
        probe.relocation.map(|r| (r.from_offset, r.target_offset)),
        row_before,
        row_after,
    );
    assert_ne!(
        probe.before.slot_count(),
        probe.after.slot_count(),
        "拆组必须改变渲染槽位数（本形状：2 → 4）"
    );
    assert_eq!(row_before, row_after, "拆组形态：锚点屏幕行不变（A20）");
}

// ── 残留缺口探针 2：锚点就是被拆开的组本身 ────────────────────────────────
#[test]
#[serial_test::serial]
fn zz_s7e_anchor_group_split_residual() {
    let mut state = browsing_state("tok");
    seed_own_card(&mut state, "s1:c1", 10, "t1", "Read");
    seed_own_card(&mut state, "s1:c2", 20, "t2", "Grep");
    seed_own_bubble(&mut state, "s1:own", 100, "below-own");
    let split = page(
        "tok",
        Some("cursor-2"),
        vec![("s1:mid", 15, vec![sub(0, user_update("between"))])],
    );

    let grid = GridSpec::grid_for(80);
    let context = context();
    let reset = crate::kit::atoms::BRIDGE_RESET_COUNTER.get();
    let mut transcript = Transcript::default();
    let mut caches: Vec<VmCacheSlot> = Vec::new();
    let first = crate::kit::atoms::VIEW_MODELS.state().read().clone();
    let before = transcript.prepare(
        &first,
        &crate::kit::atoms::TRANSCRIPT_PUBLICATION.get(),
        &mut caches,
        grid,
        grid.line_width(),
        context.clone(),
        0,
        reset,
        0,
        10,
        crate::kit::atoms::HISTORY_PREPEND_EPOCH.get(),
    );
    let before_kinds = kinds(&before);
    // 视口顶行 = 折叠组第一行（锚点身份 = Group([t1, t2])）。
    let scroll_y = before.slot_visual_start(0).expect("组起点");
    let below_row_before = before
        .slot_visual_start(slot_of_user_text(&before, "below-own"))
        .expect("气泡起点") as isize
        - scroll_y as isize;
    assert!(
        before_kinds[0].starts_with("GROUP("),
        "首屏槽位 0 必须是折叠组（实际 = {before_kinds:?}）"
    );

    dispatch_landing_page(&mut state, "tok", split);
    let second = crate::kit::atoms::VIEW_MODELS.state().read().clone();
    let after = transcript.prepare(
        &second,
        &crate::kit::atoms::TRANSCRIPT_PUBLICATION.get(),
        &mut caches,
        grid,
        grid.line_width(),
        context,
        0,
        reset,
        scroll_y,
        10,
        crate::kit::atoms::HISTORY_PREPEND_EPOCH.get(),
    );
    let relocation = transcript.take_prepend_relocation();
    let below_row_after = after
        .slot_visual_start(slot_of_user_text(&after, "below-own"))
        .expect("气泡起点") as isize
        - relocation
            .and_then(|r| r.target_for(false, scroll_y))
            .unwrap_or(scroll_y) as isize;
    println!(
        "S7E-PROBE[anchor-group-split]: 桥接槽位={} 渲染槽位={} before={:?} after={:?} relocation={:?} 下方气泡行 before={} after={}",
        state.committed.len(),
        after.slot_count(),
        before_kinds,
        kinds(&after),
        relocation.map(|r| (r.from_offset, r.target_offset)),
        below_row_before,
        below_row_after,
    );
    assert!(
        relocation.is_none(),
        "锚点身份（折叠组成员序列）在拆开后认不出 → 按设计不补偿"
    );
}
