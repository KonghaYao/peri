//! S7d 复核者探针（临时文件，复核结束即删）。
//!
//! 两个目的：
//! 1. `HISTORY_PREPEND_MARK`（桥接侧 `state.committed` 槽位坐标）与渲染侧
//!    `SlotIndex` 槽位坐标（`VIEW_MODELS.items` 下标）是否同一把尺子；
//! 2. `build_entries` 的同条起止配对形状是否安全。

use super::*;
use crate::kit::history_paging::{self, HistoryPageItem, HistoryPageResponse, HistorySubItem};
use serde_json::json;

fn sub(position: u64, update: serde_json::Value) -> HistorySubItem {
    HistorySubItem {
        position,
        update,
        system_reminder: false,
    }
}

fn tool_start(tool_id: &str, tool_name: &str) -> serde_json::Value {
    json!({
        "sessionUpdate": "tool_call",
        "toolCallId": tool_id,
        "title": tool_name,
        "rawInput": { "file_path": "/tmp/x" },
    })
}

fn tool_end(tool_id: &str, output: &str, is_error: bool) -> serde_json::Value {
    json!({
        "sessionUpdate": "tool_call_update",
        "toolCallId": tool_id,
        "rawOutput": output,
        "status": if is_error { "failed" } else { "completed" },
    })
}

fn page(
    token: &str,
    cursor: Option<&str>,
    items: Vec<(&str, u64, Vec<HistorySubItem>)>,
) -> HistoryPageResponse {
    HistoryPageResponse {
        session_id: "s1".to_string(),
        snapshot_token: token.to_string(),
        next_cursor: cursor.map(str::to_owned),
        items: items
            .into_iter()
            .map(|(entry_id, position, sub_items)| HistoryPageItem {
                entry_id: entry_id.to_string(),
                position,
                oversize: false,
                sub_items,
            })
            .collect(),
    }
}

/// 生产形状（服务端一个持久 payload 只发射起点或终点，见
/// `session_replay.rs::project_message_updates`）：两张成功工具卡分处两个条目。
/// 目的一：桥接槽位坐标 vs 渲染槽位坐标。
#[test]
#[serial]
fn zz_verify_slot_coordinates_after_tool_grouping() {
    history_paging::reset_browsing_state();
    *crate::kit::atoms::ACTIVE_SESSION_ID.state().write() = "s1".to_string();
    let mut state = make_fold_test_state();
    state.active_session_id = "s1".to_string();
    history_paging::note_initial_cursor("s1", Some("tok".to_string()), Some("c1".to_string()));

    // 继承前缀（无 position）+ 首屏自有条目（一条 user 气泡）。
    state.committed = im::Vector::from(vec![TuiRenderUnit::TuiUserBubble(TuiUserBubble::new(
        "inherited".to_string(),
    ))]);
    dispatch_for_bridge(
        &mut state,
        &AcpEventData::ReplayEntryMarked {
            entry_id: "s1:own".to_string(),
            position: 100,
        },
    );
    dispatch_and_notify(
        &mut state,
        &AcpEventData::ReplayedUserBubble {
            input_id: "i-own".to_string(),
            text: "own".to_string(),
        },
    );

    // 更早一页：起点与终点各自一个条目（真实协议形状），两张成功工具卡相邻。
    let response = page(
        "tok",
        Some("cursor-2"),
        vec![
            ("s1:old-1a", 10, vec![sub(0, tool_start("t1", "Read"))]),
            ("s1:old-1b", 11, vec![sub(0, tool_end("t1", "ok-1", false))]),
            ("s1:old-2a", 20, vec![sub(0, tool_start("t2", "Grep"))]),
            ("s1:old-2b", 21, vec![sub(0, tool_end("t2", "ok-2", false))]),
        ],
    );
    dispatch_and_notify(
        &mut state,
        &AcpEventData::HistoryPageLoaded {
            response: Box::new(response),
        },
    );

    let bridge_len = state.committed.len();
    let mark = crate::kit::atoms::HISTORY_PREPEND_MARK.get();
    let items = crate::kit::atoms::VIEW_MODELS.state().read().items.clone();
    let kinds: Vec<String> = items
        .iter()
        .map(|unit| match unit {
            TuiRenderUnit::TuiUserBubble(_) => "bubble".to_string(),
            TuiRenderUnit::TuiToolCard(card) => format!("card:{}", card.tool_id),
            TuiRenderUnit::TuiCollapsedGroup(group) => {
                format!("GROUP({})", group.view_models.len())
            }
            other => format!("{other:?}").chars().take(24).collect(),
        })
        .collect();
    panic!(
        "PROBE: 桥接 committed 槽位数 = {bridge_len}; mark（桥接槽位坐标）= {mark:?}; \
         渲染侧 VIEW_MODELS 槽位数 = {}; 渲染侧槽位序列 = {kinds:?}",
        items.len()
    );
}

/// 同一条目内「起点 + 终点」形状（`build_entries` 文档称「一页内配对」）：
/// 记录槽位用 `entries.len()`，但该条目要等本轮循环结束才 push。
#[test]
#[serial]
fn zz_verify_same_entry_tool_pair() {
    history_paging::reset_browsing_state();
    *crate::kit::atoms::ACTIVE_SESSION_ID.state().write() = "s1".to_string();
    let mut state = make_fold_test_state();
    state.active_session_id = "s1".to_string();
    history_paging::note_initial_cursor("s1", Some("tok".to_string()), Some("c1".to_string()));

    let response = page(
        "tok",
        Some("cursor-2"),
        vec![(
            "s1:old-1",
            10,
            vec![sub(0, tool_start("t1", "Read")), sub(1, tool_end("t1", "ok", false))],
        )],
    );
    dispatch_and_notify(
        &mut state,
        &AcpEventData::HistoryPageLoaded {
            response: Box::new(response),
        },
    );
    panic!(
        "PROBE: 同条起止配对未 panic；committed = {}，mark = {:?}",
        state.committed.len(),
        crate::kit::atoms::HISTORY_PREPEND_MARK.get()
    );
}

// ── [S7d 复核者探针，临时] 桥接槽位坐标 vs 渲染槽位坐标 ────────────────────
//
// 探针事实（另一份探针 `acp_events_test/zz_verify_s7d_test.rs` 实测）：
// 桥接 `state.committed` 槽位 = 4（[继承, 卡 t1, 卡 t2, 自有 bubble]），
// `HISTORY_PREPEND_MARK = Some((1, 3))`；同一次发布后渲染侧
// `VIEW_MODELS.items` 只有 3 个槽位（[继承 bubble, GROUP(2 张卡), 自有 bubble]）
// ——相邻成功工具卡被 `group_successful_tools` 压成一张折叠组。
#[test]
#[serial]
fn zz_verify_mark_slot_coordinates_against_render_slots() {
    crate::kit::atoms::init_atoms();
    paging::reset_browsing_state();
    // 渲染侧槽位 = [继承(1 行), GROUP(3 行), 自有 bubble(2 行)]，共 3 槽 6 行。
    let render_index = index_of(&[1, 3, 2]);
    // 桥接侧包络（实测值）：插入块在桥接坐标里占 [1, 3)。
    crate::kit::atoms::HISTORY_PREPEND_MARK.set(Some((1, 3)));
    let derived = super::prepend_visual_range(&render_index).expect("包络在渲染索引里取到区间");
    let delta = paging::prepend_scroll_delta(false, true, 0, 4, Some(derived));
    panic!(
        "PROBE: 派生视觉区间 = {derived:?}（插入块真实只占 1 槽 = 3 行）；\
         补偿量 delta = {delta}；正确补偿 = 3（GROUP 高度）→ 过补偿 {} 行",
        delta as isize - 3
    );
}
