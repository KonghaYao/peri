/// D1 放大性判据（复核者新增）：空闲会话的滚动触发**只认领一次**。
///
/// 覆盖实现者用例未覆盖的三点：
/// 1. 未协商（无 `snapshotToken` / `nextCursor`）时即便贴边也永不派发；
/// 2. 「滚动事件路径 + 渲染 effect 路径」在同一 tick 内交替判定、且连续 50 次
///    高频滚动（含节流窗口内 pending 尚未落地的事实）→ 通道里只允许出现一条请求；
/// 3. 结算（页返回、游标前进）之后：同页不重复派发，退避窗口过后才推进到下一页游标。
#[test]
#[serial]
fn zz_verify_d1_scroll_trigger_claims_once_and_then_advances() {
    use crate::kit::atoms::HISTORY_PAGING;
    use crate::kit::history_paging::{
        PAGE_LIMIT, request_older_page_at_boundary, request_older_page_if_due, settle_request,
    };
    crate::kit::atoms::init_atoms();
    crate::kit::history_paging::reset_browsing_state();
    drain_pages_channel();

    // (1) 未协商：视口贴边也不请求（REQ §5.3 / 判据 7）。
    assert!(
        !event::request_older_page_at_scroll_facts(false, 0, -(SCROLL_LINES as i32), 100, 20),
        "未协商分页的连接不得请求历史页"
    );
    assert!(pages_channel().lock().try_recv().is_err());

    // (2) 已协商身份：50 次高频滚动，滚动路径与渲染 effect 路径交替判定。
    crate::kit::history_paging::note_initial_cursor(
        "s1",
        Some("tok".to_string()),
        Some("c1".to_string()),
    );
    let mut dispatched = 0usize;
    for tick in 0..50 {
        let offset = 5; // 当帧 offset 仍在向前边界 2 行之外（pending 未落地）
        let pending = -(((tick % 4) + 1) as i32) * (SCROLL_LINES as i32);
        if event::request_older_page_at_scroll_facts(false, offset, pending, 100, 20) {
            dispatched += 1;
        }
        if request_older_page_at_boundary(false, offset, 100, 20) {
            dispatched += 1;
        }
    }
    assert_eq!(dispatched, 1, "同一页只允许被认领一次（并发合并）");
    let rx = pages_channel();
    let first = rx.lock().try_recv().expect("恰好一条请求");
    assert_eq!(first.session_id, "s1");
    assert_eq!(first.snapshot_token, "tok");
    assert_eq!(first.cursor, "c1");
    assert_eq!(first.limit, PAGE_LIMIT);
    assert!(rx.lock().try_recv().is_err(), "不得有第二条请求");
    let in_flight = HISTORY_PAGING
        .state()
        .read()
        .in_flight
        .clone()
        .expect("在途认领");

    // (3) 页返回、游标前进：退避窗口内不重复，过后推进到下一页。
    assert_eq!(
        settle_request(&in_flight, Ok(Some("c2".to_string()))),
        Some(Some("c2".to_string()))
    );
    assert!(
        !request_older_page_if_due(true),
        "退避窗口内不得重复派发同一页"
    );
    std::thread::sleep(std::time::Duration::from_millis(300));
    assert!(request_older_page_if_due(true), "退避过后应能继续取下一页");
    let second = rx.lock().try_recv().expect("第二条请求");
    assert_eq!(second.cursor, "c2", "下一页必须用推进后的游标");
    assert!(rx.lock().try_recv().is_err());
}

/// D6 锚点判据（复核者新增）：用真实视觉索引（`SlotIndex`）+ 生产函数
/// `prepend_scroll_delta` 量「同一可见单元在 prepend 前后的屏幕行位置」。
///
/// 屏幕行 = 该单元的视觉起始行（`slot_visual_start`）− 视口偏移（补偿后）。
/// 场景 A（插入点在视口上方，翻页常见形态）：屏幕行必须不变（A20 成立）。
/// 场景 B（插入点在视口下方 = 中间缺口填补，已知局限）：屏幕行整体上移插入高度
///（A20 不成立）——把局限量化，供后续实装锚点重定位时对照。
#[test]
fn zz_verify_d6_anchor_screen_row_before_and_after_prepend() {
    use crate::kit::history_paging::prepend_scroll_delta;
    use crate::kit::message_area::selection::{SlotIndex, SlotLines, WrappedLineInfo};
    use ratatui_kit::ratatui::text::Line;
    use std::sync::Arc;

    fn index_of(heights: &[usize]) -> SlotIndex {
        let mut index = SlotIndex::default();
        for (slot, height) in heights.iter().copied().enumerate() {
            let map = vec![WrappedLineInfo {
                logical_idx: 0,
                visual_start: 0,
                visual_end: height,
                slot_index: slot,
            }];
            index.set_slot(
                slot,
                Arc::new(SlotLines::single(Arc::new(vec![Line::from(format!(
                    "slot-{slot}"
                ))]))),
                &map,
                true,
            );
        }
        index
    }

    // ── 场景 A：更早一页插到已载入窗口顶端（视口上方）──
    // before: [继承(1) 自有末条(5)]，视口停在窗口顶端（offset 0）。
    let before = index_of(&[1, 5]);
    let own_before = before.slot_visual_start(1).expect("自有条目起始行");
    assert_eq!((own_before, before.total_visual()), (1, 6));
    // after: [继承(1) 更早页(4) 自有末条(5)] —— 插入点在视口上方。
    let after = index_of(&[1, 4, 5]);
    let own_after = after.slot_visual_start(2).expect("自有条目起始行");
    let delta = prepend_scroll_delta(false, true, before.total_visual(), after.total_visual());
    let offset_after = delta; // 生产路径：offset += delta
    assert_eq!(delta, 4, "补偿量 = 插入块高度");
    assert_eq!(
        (own_after as isize) - (offset_after as isize),
        (own_before as isize),
        "场景 A：同一可见单元的屏幕行必须保持不变（A20 成立）"
    );

    // ── 场景 B：缺口填补插到视口下方（已知局限）──
    // before: [A(4) B(4) C(4)]，vis_height 4、offset 0 → 视口只看得到 A（行 0..4）。
    let before_b = index_of(&[4, 4, 4]);
    assert_eq!((before_b.slot_visual_start(0), before_b.total_visual()), (Some(0), 12));
    // after: 缺口页(4) 插到 B 之后、C 之前 —— 插入点在视口下方（行 8 ≥ 0+4）。
    let after_b = index_of(&[4, 4, 4, 4]);
    let delta_b = prepend_scroll_delta(false, true, before_b.total_visual(), after_b.total_visual());
    let offset_after_b = delta_b;
    assert_eq!(delta_b, 4);
    assert_eq!(
        (before_b.slot_visual_start(0).unwrap() as isize) - (offset_after_b as isize),
        -4,
        "场景 B：插入点在视口下方时，原本可见的 A 被整体推出视口（A20 不成立）"
    );
}
